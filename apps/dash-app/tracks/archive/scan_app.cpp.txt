// ==== app.cpp: scan start / stop ====

double App::scanSeconds() const
{
    return scanning_ ? std::chrono::duration<double>(std::chrono::steady_clock::now() - scanT_).count() : 0;
}

void App::startScan()
{
    if (scanning_) return;
    scanRecords_ = canRecord() && !recording();
    if (scanRecords_) startRecording();
    const std::string name = recording() ? fs::path(recordingPath()).filename().string() : "scan-" + timestampName().substr(0, 15);
    {
        std::lock_guard<std::mutex> l(scanMx);
        scanner = std::make_unique<tracks::Scanner>();
        scanner->begin(name);
    }
    scanT_ = std::chrono::steady_clock::now();
    scanning_ = true;
    LOG("scan: start (%s)", name.c_str());
}

std::vector<std::string> App::stopScan()
{
    std::vector<std::string> saved;
    if (!scanning_) return saved;
    scanning_ = false;
    if (scanRecords_) stopRecording();
    scanRecords_ = false;
    std::lock_guard<std::mutex> l(scanMx);
    if (!scanner) return saved;
    std::error_code ec;
    fs::create_directories(calDataDir(), ec);
    for (const auto &[id, d] : scanner->results()) {
        if (!d.samples) continue;
        const std::string p = calDataDir() + "/" + tracks::asciiName(id) + "_scan_" + timestampName().substr(0, 15) + ".f1cal";
        std::string err;
        if (tracks::saveData(d, p, &err)) saved.push_back(p);
        LOG("scan: track %d, %llu positions, %zu edge points, %zu cells -> %s%s", id, (unsigned long long)d.samples,
            d.edges.size(), d.grid.size(), p.c_str(), err.empty() ? "" : (" failed: " + err).c_str());
    }
    LOG("scan: stop");
    return saved;
}
